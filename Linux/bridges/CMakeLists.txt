cmake_minimum_required(VERSION 3.12)
project(KinectDepthAutoLearnBridges LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)

add_executable(remold_bridge src/remold_bridge_linux.cpp)
target_link_libraries(remold_bridge PRIVATE pthread)
set_target_properties(remold_bridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")

find_path(FREENECT_INCLUDE_DIR libfreenect_sync.h PATH_SUFFIXES libfreenect)
find_library(FREENECT_LIBRARY freenect)
find_library(FREENECT_SYNC_LIBRARY freenect_sync)
if(FREENECT_INCLUDE_DIR AND FREENECT_LIBRARY AND FREENECT_SYNC_LIBRARY)
  add_executable(openkinect_bridge src/openkinect_bridge_linux.cpp)
  target_include_directories(openkinect_bridge PRIVATE ${FREENECT_INCLUDE_DIR})
  target_link_libraries(openkinect_bridge PRIVATE ${FREENECT_SYNC_LIBRARY} ${FREENECT_LIBRARY} pthread)
  set_target_properties(openkinect_bridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
else()
  message(WARNING "libfreenect not found: Remold bridge will be built; OpenKinect fallback will be skipped.")
endif()
